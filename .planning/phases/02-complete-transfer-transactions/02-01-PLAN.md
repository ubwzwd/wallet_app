**4. Add `handleTransferSubmit` async function** (before `handleSubmit`):
```typescript
const handleTransferSubmit = async () => {
  if (!validate()) return;

  const pairId = crypto.randomUUID();
  const absAmount = Math.abs(parseFloat(amount));
  const tags = tagsInput.split(',').map(t => t.trim()).filter(t => t.length > 0);
  const commonFields = {
    currency: currency.toUpperCase(),
    occurred_at: date,
    description: description.trim() || undefined,
    merchant: merchant.trim() || undefined,
    tags: tags.length > 0 ? tags : undefined,
    transfer_pair_id: pairId,
  };

  setIsSubmitting(true);

  let firstLeg: Transaction;
  try {
    firstLeg = await transactionsApi.createTransaction({
      source_id: sourceId,
      amount: -absAmount,
      ...commonFields,
    });
  } catch (err: any) {
    setIsSubmitting(false);
    const errorMessage = err.message || 'Failed to create transfer';
    if (Platform.OS === 'web') {
      window.alert(`Error\n\n${errorMessage}`);
    } else {
      Alert.alert('Error', errorMessage);
    }
    return;
  }

  try {
    await transactionsApi.createTransaction({
      source_id: destinationSourceId,
      amount: absAmount,
      ...commonFields,
    });
    queryClient.invalidateQueries({ queryKey: [QUERY_KEYS.TRANSACTIONS] });
    queryClient.invalidateQueries({ queryKey: [QUERY_KEYS.FINANCE_SOURCES, false] });
    setIsSubmitting(false);
    if (Platform.OS === 'web') {
      window.alert('Success\n\nTransfer created successfully!');
    } else {
      Alert.alert('Success', 'Transfer created successfully!');
    }
    onSuccess();
  } catch (secondErr: any) {
    try {
      await transactionsApi.deleteTransaction(firstLeg.id);
    } catch {
      setIsSubmitting(false);
      const partialMsg = 'Transfer partially created. One transaction may exist in your account \u2014 please check your transactions and delete manually if needed.';
      if (Platform.OS === 'web') {
        window.alert(`Transfer Error\n\n${partialMsg}`);
      } else {
        Alert.alert('Transfer Error', partialMsg);
      }
      return;
    }
    setIsSubmitting(false);
    const errorMessage = secondErr.message || 'Failed to create transfer';
    if (Platform.OS === 'web') {
      window.alert(`Error\n\n${errorMessage}`);
    } else {
      Alert.alert('Error', errorMessage);
    }
  }
};
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| client -> API | Transfer creation sends two POST requests with user-supplied amounts and source IDs |
| client UUID generation | `crypto.randomUUID()` generates `transfer_pair_id` client-side |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-01 | Spoofing | transfer_pair_id | accept | UUID generated client-side via `crypto.randomUUID()` (cryptographically random). Backend already validates `source_id` ownership via `current_user` check. Collision probability negligible (UUIDv4). |
| T-02-02 | Tampering | amount sign | mitigate | Client enforces `-Math.abs()` for debit and `+Math.abs()` for credit. Backend already validates `source_id` belongs to `current_user` (owner check in transactions.py). Amount sign convention is display-only concern. |
| T-02-03 | Denial of Service | sequential POSTs | accept | Two sequential POST calls per transfer creation is acceptable — no amplification vector. Backend rate limiting deferred to SEC-01 (v2). |
| T-02-04 | Information Disclosure | partial failure message | accept | Error message reveals that a partial transaction may exist but includes no sensitive data (no IDs, amounts, or source names in the alert text). |
| T-02-05 | Elevation of Privilege | cross-source transfer | mitigate | Backend `create_transaction` handler validates `source_id` belongs to `current_user` (owner check). Client cannot create transactions against other users' sources. |
</threat_model>
